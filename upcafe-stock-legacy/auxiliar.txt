<?php

$sql = "SELECT id, nombre, cantidad FROM Productos";
$result = $conexion->query($sql);

    echo "<h2>Lista de Productos</h2>";
    echo "<table border='2' cellpadding='6' cellspacing='3'>";
    echo "<tr><th>ID</th><th>Nombre</th><th>Cantidad</th></tr>";
    while($row = $result->fetch_assoc()) {
        echo "<tr>";
        echo "<td>" . $row["id"] . "</td>";
        echo "<td>" . $row["nombre"] . "</td>";
        echo "<td>" . $row["cantidad"] . "</td>";
        echo "</tr>";
    }
    echo "</table>";
} else {
    echo "No hay productos cargados.";
}

$conexion->close();
?>

    $producto = $resultado->fetch_assoc();
}

// Paso 2: actualizar producto
if (isset($_POST['modificar'])) {
    $id = intval($_POST['id']);
    $nombre = $_POST['nombre'];
    $cantidad = intval($_POST['cantidad']);

    $sql = "UPDATE Productos SET nombre='$nombre', cantidad=$cantidad WHERE id=$id";
    if ($conexion->query($sql) === TRUE) {
        echo "Producto modificado con éxito.";

        echo "Error al modificar: " . $conexion->error;
